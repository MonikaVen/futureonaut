Level 7 — PyTorch mode

Connect what you saw in the game to real deep-learning code.

import torch
import torch.nn as nn
X = torch.tensor([...])
y = torch.tensor([...])
model = nn.Sequential(
nn.Linear(3, 3),
nn.ReLU(),
nn.Linear(3, 1),
nn.Sigmoid()
)
loss_fn = nn.BCELoss()
optimizer = torch.optim.SGD(model.parameters(), lr=0.05)
for epoch in range(100):
prediction = model(X)
loss = loss_fn(prediction, y)
optimizer.zero_grad()
loss.backward()
optimizer.step()
